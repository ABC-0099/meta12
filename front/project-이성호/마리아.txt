1. 부서별 직원 수 구하기
select department, count(*)
from employees
group by department;

2. 부서별 급여 합계와 평균 구하기
select department,
sum(salary),avg(salary)
from employees
group by department;

3. 부서별 급여 합계와 평균 구하고, 평균이 4000이상인 것만 출력
select department,
sum(salary),avg(salary)
from employees
group by department
having avg(salary) >= 4000;

4. 부서별, 직급별 직원 수 구하기
select department,role, count(*)
from employees
group by department, role;

-- 부서별 사원 수
select department_id, count(*)
from employees
group by department_id;

(
    select base_salary
    from salaries s
    where s.employee_id = e.employee_id
)as salary
from employees e
order by salary desc
limit 1;

-- 카테고리별 주문 건수 구하기
select category, count(*) 
from orders
group by category;

select category 카테고리, count(*) 주문건수
from orders
group by category;


-- 카테고리별 총 매출액과 평균 단가 구하기
select category, sum(price*quantity) 총매출액, avg(price) 평균단가
from orders
group by category;

-- 총 매출액이 100,000원 이상인 카테고리만 출력
select category, sum(price*quantity) 총매출액
from orders
group by category
having sum(price*quantity) >= 100000;

--주문일자별로 카테고리별로 일일매출액 출력
select order_date 주문일자,category 카테고리, sum(price*quantity) 일일매출액
from orders
group by order_date,category;
